version: "3.9"

name: kafka-stack

services:
  kafka-1:
    image: apache/kafka:4.1.0
    container_name: kafka-1
    hostname: kafka-1
    ports:
      - "0.0.0.0:9092:9092"
    environment:
      KAFKA_NODE_ID: 1
      KAFKA_PROCESS_ROLES: "broker,controller"

      KAFKA_LISTENERS: "INTERNAL://:29092,EXTERNAL://:9092,CONTROLLER://:9093"
      KAFKA_ADVERTISED_LISTENERS: "INTERNAL://kafka-1:29092,EXTERNAL://10.172.9.189:9092"

      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: "INTERNAL:PLAINTEXT,EXTERNAL:PLAINTEXT,CONTROLLER:PLAINTEXT"
      KAFKA_INTER_BROKER_LISTENER_NAME: "INTERNAL"
      KAFKA_CONTROLLER_LISTENER_NAMES: "CONTROLLER"
      KAFKA_CONTROLLER_QUORUM_VOTERS: "1@kafka-1:9093,2@kafka-2:9093,3@kafka-3:9093"

      KAFKA_CFG_AUTO_CREATE_TOPICS_ENABLE: "false"
      KAFKA_LOG_DIRS: "/var/lib/kafka/data"

      # ✅ Make this stable (set it in .env or hardcode)
      CLUSTER_ID: "${KAFKA_CLUSTER_ID}"

2- Minio docker 


--------------------------------------------------

services:
  minio:
    image: minio/minio:RELEASE.2025-01-20T14-49-07Z
    container_name: room-minio
    command: server /data --console-address ":9001"

    environment:
      MINIO_ROOT_USER: minioadmin
      MINIO_ROOT_PASSWORD: minioadmin
      TZ: Asia/Phnom_Penh
      MINIO_SERVER_URL: "http://10.172.9.189:9000"
      MINIO_BROWSER_REDIRECT_URL: "http://10.172.9.189:9001"

    ports:
      - "0.0.0.0:9000:9000"
      - "0.0.0.0:9001:9001"

    volumes:
      - room_minio_data:/data

    # ✅ Healthcheck that works with minio image
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9000/minio/health/live"]
      interval: 10s
      timeout: 5s
      retries: 20

    restart: unless-stopped

  minio-mc:
    image: minio/mc:RELEASE.2025-07-21T05-28-08Z
    container_name: room-minio-mc
    depends_on:
      minio:
        condition: service_healthy
    entrypoint: >
      /bin/sh -c "
      mc alias set local http://minio:9000 minioadmin minioadmin &&
      mc mb -p local/room-media || true &&
      mc anonymous set download local/room-media || true &&
      echo 'MinIO bucket ready: room-media' &&
      exit 0
      "
    restart: "no"

volumes:
  room_minio_data:
